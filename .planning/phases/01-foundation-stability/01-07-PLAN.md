@asynccontextmanager
async def lifespan(app: FastAPI):
    # Startup validation
    print("=" * 50)
    print("Filharmonia AI - Starting up...")
    print("=" * 50)

    # Check PyTorch and GPU
    try:
        import torch
        gpu_available = torch.cuda.is_available()
        device = "CUDA" if gpu_available else "CPU"
        if gpu_available:
            gpu_name = torch.cuda.get_device_name(0)
            print(f"  PyTorch device: {device} ({gpu_name})")
        else:
            print(f"  PyTorch device: {device}")
        print(f"  PyTorch version: {torch.__version__}")
    except ImportError:
        print("  WARNING: PyTorch not installed")

    # Check torchaudio
    try:
        import torchaudio
        print(f"  torchaudio version: {torchaudio.__version__}")
        print(f"  Audio backend: torchaudio (native)")
    except ImportError:
        print("  WARNING: torchaudio not installed, falling back to librosa")

    # Check librosa and its backends
    try:
        import librosa
        print(f"  librosa version: {librosa.__version__}")

        # Check soundfile (preferred backend)
        try:
            import soundfile
            print(f"  Audio backend: soundfile {soundfile.__version__} (fast)")
        except ImportError:
            print("  WARNING: soundfile not installed")

            # Check audioread fallback
            try:
                import audioread
                print(f"  Audio backend: audioread (slow, ffmpeg-based)")
                print("  RECOMMENDATION: Install soundfile for faster audio loading")
            except ImportError:
                print("  ERROR: No audio backend available!")
                print("  Install soundfile: pip install soundfile")
    except ImportError:
        print("  WARNING: librosa not installed")

    # Check ffmpeg (needed for some audio formats)
    import subprocess
    try:
        result = subprocess.run(['ffmpeg', '-version'], capture_output=True, text=True)
        if result.returncode == 0:
            version_line = result.stdout.split('\n')[0]
            print(f"  ffmpeg: {version_line}")
        else:
            print("  WARNING: ffmpeg not working correctly")
    except FileNotFoundError:
        print("  WARNING: ffmpeg not found in PATH (needed for some audio formats)")

    print("=" * 50)
    print("Startup complete!")
    print("=" * 50)

    yield  # Application runs here

    # Shutdown
    print("Graceful shutdown: marking active analysis jobs as interrupted...")
    # ... rest of existing shutdown code
```

The requirements.txt already has pinned versions:
```
torch==2.5.1+cu121
torchaudio==2.5.1+cu121
torchvision==0.20.1+cu121
```

This is correct - the `+cu121` suffix pins to CUDA 12.1.

Add a comment explaining the pinning strategy at the top of the file or near the PyTorch entries:

```
# ====================
# PyTorch with CUDA 12.1
# ====================
# Pinned to specific CUDA version for reproducibility.
# For CPU-only: pip install torch==2.5.1 torchaudio==2.5.1 torchvision==0.20.1
# For CUDA 11.8: pip install torch==2.5.1+cu118 --index-url https://download.pytorch.org/whl/cu118
# For ROCm 6.0: pip install torch==2.5.1+rocm6.0 --index-url https://download.pytorch.org/whl/rocm6.0
torch==2.5.1+cu121
torchaudio==2.5.1+cu121
torchvision==0.20.1+cu121
```

Also verify the versions are consistent (torch, torchaudio, torchvision should all be same base version).

Note: The `+cuXXX` suffix packages are hosted on PyTorch's custom index, not PyPI. The current requirements.txt may need adjustment if installing fresh.

For production, consider creating:
1. `requirements-cuda121.txt` for CUDA 12.1
2. `requirements-cpu.txt` for CPU-only
3. `requirements-rocm.txt` for AMD GPUs

For now, just add the comment documentation.

@app.get("/health")
async def health():
    """Health check with system information"""
    import torch

    # Determine audio backend
    audio_backend = "unknown"
    try:
        import soundfile
        audio_backend = f"soundfile {soundfile.__version__}"
    except ImportError:
        try:
            import audioread
            audio_backend = "audioread (ffmpeg)"
        except ImportError:
            audio_backend = "none"

    return {
        "status": "healthy",
        "device": "cuda" if torch.cuda.is_available() else "cpu",
        "audio_backend": audio_backend,
        "torch_version": torch.__version__
    }
```

This allows programmatic checking of the audio backend without parsing logs.
